Good call — MigLayout makes this much cleaner than GridBagLayout. Here's the same dashboard rebuilt with MigLayout throughout.

Reminder — make sure this is in your `pom.xml`:

```xml
<dependency>
    <groupId>com.miglayout</groupId>
    <artifactId>miglayout-swing</artifactId>
    <version>11.4.2</version>
</dependency>
```

## 1. ui/MainFrame.java — MigLayout root

```java
package com.example.dashboard.ui;

import com.example.dashboard.pages.DashboardPage;
import com.example.dashboard.pages.StudentsPage;
import com.example.dashboard.pages.TeachersPage;
import net.miginfocom.swing.MigLayout;

import javax.swing.*;
import java.awt.*;

public class MainFrame extends JFrame {

    private final CardLayout cardLayout = new CardLayout();
    private final JPanel contentPanel = new JPanel(cardLayout);

    public MainFrame() {
        setTitle("Education Dashboard");
        setSize(1200, 800);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);

        // Two columns: sidebar fixed at 220px, content grows to fill the rest
        setLayout(new MigLayout("insets 0, gap 0, fill", "[220!][grow]", "[grow]"));

        SidebarPanel sidebar = new SidebarPanel(this::showPage);
        add(sidebar, "grow");

        contentPanel.add(new DashboardPage(), "Dashboard");
        contentPanel.add(new StudentsPage(), "Students");
        contentPanel.add(new TeachersPage(), "Teachers");
        add(contentPanel, "grow");

        showPage("Dashboard");
    }

    private void showPage(String name) {
        cardLayout.show(contentPanel, name);
    }
}
```

## 2. ui/SidebarPanel.java — MigLayout, one column, wrap

```java
package com.example.dashboard.ui;

import net.miginfocom.swing.MigLayout;

import javax.swing.*;
import java.awt.*;
import java.util.function.Consumer;

public class SidebarPanel extends JPanel {

    private final Consumer<String> onMenuSelected;

    public SidebarPanel(Consumer<String> onMenuSelected) {
        this.onMenuSelected = onMenuSelected;

        // "wrap 1" = every added component goes on its own row
        setLayout(new MigLayout("insets 20 10 20 10, wrap 1, gap 4", "[grow]"));
        setBackground(new Color(0x0F1E5B));

        addMenuButton("Dashboard");
        addMenuButton("Students");
        addMenuButton("Teachers");
    }

    private void addMenuButton(String name) {
        JButton button = new JButton(name);
        button.setFocusPainted(false);
        button.setHorizontalAlignment(SwingConstants.LEFT);
        button.setForeground(Color.WHITE);
        button.setBackground(new Color(0x0F1E5B));
        button.setBorder(BorderFactory.createEmptyBorder(10, 16, 10, 16));
        button.setFont(new Font("Segoe UI", Font.PLAIN, 14));

        button.addMouseListener(new java.awt.event.MouseAdapter() {
            public void mouseEntered(java.awt.event.MouseEvent e) {
                button.setBackground(new Color(0x1B2E7A));
            }
            public void mouseExited(java.awt.event.MouseEvent e) {
                button.setBackground(new Color(0x0F1E5B));
            }
        });

        button.addActionListener(e -> onMenuSelected.accept(name));

        add(button, "growx, height 40!");
    }
}
```

## 4. ui/StatCardPanel.java — MigLayout

```java
package com.example.dashboard.ui;

import net.miginfocom.swing.MigLayout;

import javax.swing.*;
import java.awt.*;

public class StatCardPanel extends JPanel {

    public StatCardPanel(String title, String number, Color badgeColor) {
        setLayout(new MigLayout("insets 18, gap 14", "[pref!][grow]"));
        setBackground(Color.WHITE);
        setBorder(BorderFactory.createLineBorder(new Color(0xE5E7EB), 1, true));

        JPanel badge = new JPanel() {
            @Override
            protected void paintComponent(Graphics g) {
                Graphics2D g2 = (Graphics2D) g.create();
                g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
                g2.setColor(badgeColor);
                g2.fillOval(0, 0, getWidth(), getHeight());
                g2.dispose();
            }
        };
        badge.setOpaque(false);
        add(badge, "width 48!, height 48!");

        JPanel textPanel = new JPanel(new MigLayout("insets 0, wrap 1, gap 2"));
        textPanel.setOpaque(false);

        JLabel titleLabel = new JLabel(title);
        titleLabel.setFont(new Font("Segoe UI", Font.PLAIN, 14));
        titleLabel.setForeground(Color.GRAY);

        JLabel numberLabel = new JLabel(number);
        numberLabel.setFont(new Font("Segoe UI", Font.BOLD, 26));

        textPanel.add(titleLabel);
        textPanel.add(numberLabel);
        add(textPanel);
    }
}
```

```java
package com.example.dashboard.ui;

import net.miginfocom.swing.MigLayout;

import javax.swing.*;
import java.awt.*;

public class DashboardCard extends JPanel {

    private final JPanel bodyPanel = new JPanel(new BorderLayout());

    public DashboardCard(String title) {
        setLayout(new MigLayout("insets 16 18 16 18, wrap 1, fill", "[grow]", "[pref!][grow]"));
        setBackground(Color.WHITE);
        setBorder(BorderFactory.createLineBorder(new Color(0xE5E7EB), 1, true));

        JLabel titleLabel = new JLabel(title);
        titleLabel.setFont(new Font("Segoe UI", Font.BOLD, 16));
        add(titleLabel, "growx");

        bodyPanel.setOpaque(false);
        add(bodyPanel, "grow, gaptop 10");
    }

    public void setBody(JComponent content) {
        bodyPanel.removeAll();
        bodyPanel.add(content, BorderLayout.CENTER);
        bodyPanel.revalidate();
        bodyPanel.repaint();
    }
}
```

## 7. ui/EventCalendarPanel.java — MigLayout, 7-column grid

```java
package com.example.dashboard.ui;

import net.miginfocom.swing.MigLayout;

import javax.swing.*;
import java.awt.*;
import java.time.LocalDate;
import java.time.YearMonth;

public class EventCalendarPanel extends JPanel {

    public EventCalendarPanel() {
        setLayout(new MigLayout("insets 0, wrap 1, gapy 10", "[grow]"));
        setOpaque(false);

        YearMonth month = YearMonth.now();
        JLabel monthLabel = new JLabel(month.getMonth() + " " + month.getYear());
        monthLabel.setFont(new Font("Segoe UI", Font.BOLD, 14));
        add(monthLabel);

        // 7 equal-width columns for the day grid
        JPanel grid = new JPanel(new MigLayout("insets 0, wrap 7, gap 4 6", "[grow,fill]7"));
        grid.setOpaque(false);

        for (String d : new String[]{"SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"}) {
            JLabel dLabel = new JLabel(d, SwingConstants.CENTER);
            dLabel.setFont(new Font("Segoe UI", Font.BOLD, 11));
            dLabel.setForeground(Color.GRAY);
            grid.add(dLabel, "growx");
        }

        int startDayOfWeek = month.atDay(1).getDayOfWeek().getValue() % 7; // Sun = 0
        for (int i = 0; i < startDayOfWeek; i++) grid.add(new JLabel(""), "growx");

        int today = LocalDate.now().getDayOfMonth();
        for (int day = 1; day <= month.lengthOfMonth(); day++) {
            JLabel dayLabel = new JLabel(String.valueOf(day), SwingConstants.CENTER);
            if (day == today) {
                dayLabel.setOpaque(true);
                dayLabel.setBackground(new Color(0x0F1E5B));
                dayLabel.setForeground(Color.WHITE);
            }
            grid.add(dayLabel, "growx, height 26!");
        }

        add(grid, "growx");
    }
}
```

## 8. pages/DashboardPage.java — MigLayout for the whole grid

This replaces the `GridBagLayout` version entirely:

```java
package com.example.dashboard.pages;

import com.example.dashboard.ui.*;
import net.miginfocom.swing.MigLayout;

import javax.swing.*;
import java.awt.*;

public class DashboardPage extends JPanel {

    public DashboardPage() {
        setLayout(new BorderLayout());
        setBackground(new Color(0xF5F6FA));

        add(new TopBarPanel(), BorderLayout.NORTH);

        // 3 equal columns, rows sized: stat row fixed height, chart rows grow
        JPanel body = new JPanel(new MigLayout(
                "insets 0 20 20 20, gap 16, fill",
                "[grow][grow][grow]",
                "[pref!][grow][grow]"
        ));
        body.setOpaque(false);

        // Row 1: 3 stat cards
        body.add(new StatCardPanel("Students", "20000", new Color(0x2DD4BF)), "grow");
        body.add(new StatCardPanel("Teachers", "3000", new Color(0x3B82F6)), "grow");
        body.add(new StatCardPanel("Parents", "5500", new Color(0xF59E0B)), "grow, wrap");

        // Row 2: donut card (spans 2 cols) | notice board card
        DashboardCard studentsCard = new DashboardCard("Students");
        studentsCard.setBody(new StudentsDonutPanel());

        DashboardCard noticeCard = new DashboardCard("Notice Board");
        noticeCard.setBody(new NoticeBoardPanel());

        body.add(studentsCard, "span 2, grow");
        body.add(noticeCard, "grow, wrap");

        // Row 3: calendar card (spans 2 cols) | expense chart card
        DashboardCard calendarCard = new DashboardCard("Event Calendar");
        calendarCard.setBody(new EventCalendarPanel());

        DashboardCard expenseCard = new DashboardCard("Expense");
        expenseCard.setBody(new ExpenseChartPanel());

        body.add(calendarCard, "span 2, grow");
        body.add(expenseCard, "grow");

        add(new JScrollPane(body), BorderLayout.CENTER);
    }
}
```

## Key MigLayout syntax you're using here

- `"insets 0, fill"` — no outer padding, children fill the space.
- `"[220!][grow]"` — column definitions: `220!` = fixed 220px, `grow` = takes remaining space.
- `"wrap N"` — after every N components added, start a new row automatically (no manual `gridx`/`gridy`).
- `"span 2"` — this component spans 2 columns (used for the donut and calendar cards, matching the wider boxes in your screenshot).
- `"grow"` on a component — let it expand to fill its cell, both directions.
- `"growx"` — expand horizontally only.

`StudentsDonutPanel` and `ExpenseChartPanel` (the JFreeChart ones) don't need changes — they still use `BorderLayout` internally since MigLayout doesn't add value inside a `ChartPanel`.
