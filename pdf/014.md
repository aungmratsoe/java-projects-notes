```java
package com.ams.report.chart;

import com.ams.report.model.MonthlyStudentData;
import com.ams.report.model.StudentReportData;

import java.util.List;

import org.jfree.chart.ChartFactory;
import org.jfree.chart.JFreeChart;

public final class StudentChartFactory {

    private StudentChartFactory() {
    }

    public static JFreeChart createMonthlyChart(
            StudentReportData data) {

        DefaultCategoryDataset dataset =
                new DefaultCategoryDataset();

        List<MonthlyStudentData> monthlyData =
                data.getMonthlyData();

        for (MonthlyStudentData item : monthlyData) {

                    item.getCount(),
                    "Students",
                    item.getMonth()
            );
        }

        return ChartFactory.createBarChart(
                "Monthly Student Registration",
                "Month",
                "Students",
                dataset
        );
    }
}
```

# 8. Vector Chart PDF

```java
package com.ams.report.pdf;

import com.orsonpdf.PDFDocument;
import com.orsonpdf.PDFGraphics2D;
import com.orsonpdf.Page;

import java.awt.Rectangle;
import java.io.File;
import java.io.FileOutputStream;
import java.io.IOException;

import org.jfree.chart.JFreeChart;

public final class VectorChartPdf {

    private VectorChartPdf() {
    }

    public static void export(
            JFreeChart chart,
            File outputFile,
            int width,
            int height
    ) throws IOException {

        PDFDocument pdf =
                new PDFDocument();

        Page page =
                pdf.createPage(
                    new Rectangle(
                        width,
                        height
                    )
                );

        PDFGraphics2D graphics =
                page.getGraphics2D();

        chart.draw(
                graphics,
                new Rectangle(
                    0,
                    0,
                    width,
                    height
                )
        );

        try (FileOutputStream output =
                     new FileOutputStream(
                         outputFile
                     )) {

            output.write(
                    pdf.getPDFBytes()
            );
        }
    }
}
```

# 9. `PdfLayout`

ဒီ class က architecture ရဲ့ အရေးကြီးဆုံး class တစ်ခုပါ။

```java
package com.ams.report.pdf;

import java.io.IOException;

import org.apache.pdfbox.pdmodel.PDDocument;
import org.apache.pdfbox.pdmodel.PDPage;
import org.apache.pdfbox.pdmodel.PDPageContentStream;
import org.apache.pdfbox.pdmodel.common.PDRectangle;

public class PdfLayout {

    private final PDDocument document;

    private final float pageWidth;
    private final float pageHeight;

    private final float margin;

    private PDPage page;
    private PDPageContentStream content;

    private float currentY;

    private int pageNumber;

    public PdfLayout(
            PDDocument document,
            PDRectangle pageSize,
            float margin
    ) throws IOException {

        this.document = document;

        this.pageWidth =
                pageSize.getWidth();

        this.pageHeight =
                pageSize.getHeight();

        this.margin = margin;

        newPage();
    }

    public void newPage()
            throws IOException {

        if (content != null) {
            content.close();
        }

        page =
                new PDPage(
                    new PDRectangle(
                        pageWidth,
                        pageHeight
                    )
                );

        document.addPage(page);

        content =
                new PDPageContentStream(
                    document,
                    page
                );

        pageNumber++;

        currentY =
                pageHeight - margin;
    }

    public boolean hasSpace(
            float height
    ) {

        return currentY - height
                >= margin;
    }

    public void requireSpace(
            float height
    ) throws IOException {

        if (!hasSpace(height)) {
            newPage();
        }
    }

    public float getX() {
        return margin;
    }

    public float getY() {
        return currentY;
    }

    public float getPageWidth() {
        return pageWidth;
    }

    public float getPageHeight() {
        return pageHeight;
    }

    public float getContentWidth() {
        return pageWidth - (margin * 2);
    }

    public PDPage getPage() {
        return page;
    }

    public PDPageContentStream getContent() {
        return content;
    }

    public int getPageNumber() {
        return pageNumber;
    }

    public void moveDown(
            float height
    ) {

        currentY -= height;
    }

    public void close()
            throws IOException {

        if (content != null) {
            content.close();
        }
    }
}
```

# 11. `PdfTable`

ဒီ version က pagination support ပါမယ်။

```java
package com.ams.report.pdf;

import java.io.IOException;
import java.util.List;

import org.apache.pdfbox.pdmodel.PDPageContentStream;
import org.apache.pdfbox.pdmodel.font.PDFont;

public class PdfTable {

    private final String[] headers;
    private final List<String[]> rows;

    private final float[] columnWidths;

    private final float rowHeight;

    public PdfTable(
            String[] headers,
            List<String[]> rows,
            float[] columnWidths,
            float rowHeight) {

        this.headers = headers;
        this.rows = rows;
        this.columnWidths = columnWidths;
        this.rowHeight = rowHeight;
    }

    public void draw(
            PdfLayout layout,
            PDFont headerFont,
            PDFont bodyFont,
            float fontSize
    ) throws IOException {

        drawHeader(
                layout,
                headerFont,
                fontSize
        );

        for (String[] row : rows) {

            if (!layout.hasSpace(rowHeight)) {

                layout.newPage();

                drawHeader(
                        layout,
                        headerFont,
                        fontSize
                );
            }

            drawRow(
                    layout,
                    bodyFont,
                    fontSize,
                    row
            );
        }
    }

    private void drawHeader(
            PdfLayout layout,
            PDFont font,
            float fontSize
    ) throws IOException {

        drawCells(
                layout,
                font,
                fontSize,
                headers
        );

        layout.moveDown(rowHeight);
    }

    private void drawRow(
            PdfLayout layout,
            PDFont font,
            float fontSize,
            String[] values
    ) throws IOException {

        drawCells(
                layout,
                font,
                fontSize,
                values
        );

        layout.moveDown(rowHeight);
    }

    private void drawCells(
            PdfLayout layout,
            PDFont font,
            float fontSize,
            String[] values
    ) throws IOException {

        PDPageContentStream content =
                layout.getContent();

        float x = layout.getX();

        float y = layout.getY();

        for (int i = 0;
             i < columnWidths.length;
             i++) {

            float width =
                    columnWidths[i];

            content.addRect(
                    x,
                    y - rowHeight,
                    width,
                    rowHeight
            );

            content.stroke();

            content.beginText();

            content.setFont(
                    font,
                    fontSize
            );

            content.newLineAtOffset(
                    x + 5,
                    y - 16
            );

            content.showText(
                    values[i]
            );

            content.endText();

            x += width;
        }
    }
}
```

# 12. အခု `StudentReportPdfExporter`

ဒီ class က အားလုံးကို coordinate လုပ်မယ်။

```java
package com.ams.report.pdf;

import com.ams.report.chart.StudentChartFactory;
import com.ams.report.model.StudentReportData;

import java.io.File;
import java.io.IOException;
import java.util.List;

import org.apache.pdfbox.multipdf.LayerUtility;
import org.apache.pdfbox.pdmodel.PDDocument;
import org.apache.pdfbox.pdmodel.PDPage;
import org.apache.pdfbox.pdmodel.PDRectangle;
import org.apache.pdfbox.pdmodel.font.PDType1Font;
import org.apache.pdfbox.pdmodel.graphics.form.PDFormXObject;

import org.jfree.chart.JFreeChart;

public final class StudentReportPdfExporter {

    private StudentReportPdfExporter() {
    }

    public static void export(
            StudentReportData data,
            File outputFile
    ) throws IOException {

        try (PDDocument document =
                     new PDDocument()) {

            PdfLayout layout =
                    new PdfLayout(
                        document,
                        PDRectangle.A4,
                        40
                    );

            drawHeader(
                    layout
            );

            drawSummary(
                    layout,
                    data
            );

            File chartFile =
                    File.createTempFile(
                        "student-chart-",
                        ".pdf"
                    );

            try {

                JFreeChart chart =
                        StudentChartFactory
                                .createMonthlyChart(
                                    data
                                );

                VectorChartPdf.export(
                        chart,
                        chartFile,
                        700,
                        300
                );

                drawChart(
                        document,
                        layout,
                        chartFile
                );

            } finally {

                chartFile.delete();
            }

            drawTable(
                    layout,
                    data
            );

            drawFooter(
                    layout
            );

            layout.close();

            document.save(
                    outputFile
            );
        }
    }

    private static void drawHeader(
            PdfLayout layout
    ) throws IOException {

        layout.requireSpace(40);

        PdfText.draw(
                layout.getContent(),
                "STUDENT MANAGEMENT SYSTEM",
                layout.getX(),
                layout.getY(),
                PDType1Font.HELVETICA_BOLD,
                20
        );

        layout.moveDown(35);

        PdfText.draw(
                layout.getContent(),
                "Student Registration Report",
                layout.getX(),
                layout.getY(),
                PDType1Font.HELVETICA,
                12
        );

        layout.moveDown(30);
    }

    private static void drawSummary(
            PdfLayout layout,
            StudentReportData data
    ) throws IOException {

        layout.requireSpace(60);

        PdfText.draw(
                layout.getContent(),
                "Total Students: "
                        + data.getTotalStudents(),
                layout.getX(),
                layout.getY(),
                PDType1Font.HELVETICA,
                11
        );

        layout.moveDown(18);

        PdfText.draw(
                layout.getContent(),
                "Male: "
                        + data.getMaleStudents(),
                layout.getX(),
                layout.getY(),
                PDType1Font.HELVETICA,
                11
        );

        layout.moveDown(18);

        PdfText.draw(
                layout.getContent(),
                "Female: "
                        + data.getFemaleStudents(),
                layout.getX(),
                layout.getY(),
                PDType1Font.HELVETICA,
                11
        );

        layout.moveDown(30);
    }

    private static void drawChart(
            PDDocument document,
            PdfLayout layout,
            File chartFile
    ) throws IOException {

        float chartHeight = 240;

        layout.requireSpace(
                chartHeight
        );

        try (
            PDDocument chartDocument =
                    PDDocument.load(chartFile)
        ) {

            LayerUtility utility =
                    new LayerUtility(
                        document
                    );

            PDPage page =
                    layout.getPage();

            PDFormXObject form =
                    utility.importPageAsForm(
                        chartDocument,
                        0
                    );

            utility.appendFormAsLayer(
                    page,
                    form,
                    new java.awt.geom.AffineTransform(
                        0.7,
                        0,
                        0,
                        0.7,
                        layout.getX(),
                        layout.getY()
                                - chartHeight
                    ),
                    "Student Chart"
            );
        }

        layout.moveDown(
                chartHeight + 20
        );
    }

    private static void drawTable(
            PdfLayout layout,
            StudentReportData data
    ) throws IOException {

        String[] headers = {
            "ID",
            "Name",
            "Gender",
            "Course"
        };

        List<String[]> rows =
                data.getStudents()
                    .stream()
                    .map(student ->
                        student.toArray()
                    )
                    .toList();

        float[] widths = {
            70,
            190,
            90,
            160
        };

        PdfTable table =
                new PdfTable(
                    headers,
                    rows,
                    widths,
                    25
                );

        table.draw(
                layout,
                PDType1Font.HELVETICA_BOLD,
                PDType1Font.HELVETICA,
                9
        );
    }

    private static void drawFooter(
            PdfLayout layout
    ) throws IOException {

        PdfText.draw(
                layout.getContent(),
                "Student Management System",
                layout.getX(),
                20,
                PDType1Font.HELVETICA,
                8
        );

        PdfText.draw(
                layout.getContent(),
                "Page "
                        + layout.getPageNumber(),
                layout.getPageWidth()
                        - 70,
                20,
                PDType1Font.HELVETICA,
                8
        );
    }
}
```
