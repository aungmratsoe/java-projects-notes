```java
public class StudentChartService {

    public JFreeChart createGenderChart(
            StudentReportData data) {

        DefaultPieDataset dataset =
                new DefaultPieDataset();

        dataset.setValue(
                "Male",
                data.getMaleStudents()
        );

        dataset.setValue(
                "Female",
                data.getFemaleStudents()
        );

        return ChartFactory.createPieChart(
                "Student Gender Distribution",
                dataset,
                true,
                true,
                false
        );
    }
}
```

# 13. `PdfText` utility

```java
public class PdfText {

    public static void drawText(
            PDPageContentStream content,
            String text,
            float x,
            float y,
            float fontSize
    ) throws IOException {

        content.beginText();

        content.setFont(
                PDType1Font.HELVETICA,
                fontSize
        );

        content.newLineAtOffset(x, y);

        content.showText(text);

        content.endText();
    }
}
```

အခု—

```java
PdfText.drawText(
        content,
        "STUDENT MANAGEMENT SYSTEM",
        40,
        550,
        20
);
```
