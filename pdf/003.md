# 7. `drawCard()` Complete Version

```java
private void drawCard(
        PDDocument document,
        PDPageContentStream content,
        Student student,
        float x,
        float y)
        throws IOException {

    // =========================
    // Background
    // =========================

    content.setNonStrokingColor(
            245,
            247,
            250
    );

    content.addRect(
            x,
            y,
            CARD_WIDTH,
            CARD_HEIGHT
    );

    content.fill();


    // =========================
    // Border
    // =========================

    content.setStrokingColor(
            40,
            40,
            40
    );

    content.setLineWidth(1);

    content.addRect(
            x + 1,
            y + 1,
            CARD_WIDTH - 2,
            CARD_HEIGHT - 2
    );

    content.stroke();


    // =========================
    // University
    // =========================

    drawText(
            content,
            "ABC UNIVERSITY",
            x + 15,
            y + 125,
            14
    );


    // =========================
    // Photo
    // =========================

    File photoFile =
            new File(student.getPhotoPath());

    if (photoFile.exists()) {

        PDImageXObject photo =
                PDImageXObject.createFromFile(
                        photoFile.getAbsolutePath(),
                        document
                );

        content.drawImage(
                photo,
                x + 15,
                y + 45,
                65,
                65
        );
    }


    // =========================
    // Student Information
    // =========================

    drawText(
            content,
            "Student ID : "
                    + student.getStudentId(),
            x + 90,
            y + 90,
            9
    );

    drawText(
            content,
            "Name : "
                    + student.getName(),
            x + 90,
            y + 75,
            9
    );

    drawText(
            content,
            "Major : "
                    + student.getSpecialization(),
            x + 90,
            y + 60,
            9
    );

    drawText(
            content,
            "Phone : "
                    + student.getPhone(),
            x + 90,
            y + 45,
            9
    );


    // =========================
    // QR Code
    // =========================

    File qrFile =
            new File(student.getQrPath());

    if (qrFile.exists()) {

        PDImageXObject qr =
                PDImageXObject.createFromFile(
                        qrFile.getAbsolutePath(),
                        document
                );

        content.drawImage(
                qr,
                x + 175,
                y + 5,
                50,
                50
        );
    }
}
```

# 8. A4 PDF Generator

အခု A4 document တစ်ခု create မယ်။

```java
public void generateA4(
        List<Student> students,
        File outputFile)
        throws IOException {

    try (PDDocument document = new PDDocument()) {

        PDPage page =
                new PDPage(PDRectangle.A4);

        document.addPage(page);

        try (PDPageContentStream content =
                     new PDPageContentStream(
                             document,
                             page
                     )) {

            for (int i = 0;
                 i < students.size() && i < 4;
                 i++) {

                float x;
                float y;

                if (i == 0) {
                    x = 50;
                    y = 600;
                }
                else if (i == 1) {
                    x = 300;
                    y = 600;
                }
                else if (i == 2) {
                    x = 50;
                    y = 400;
                }
                else {
                    x = 300;
                    y = 400;
                }

                drawCard(
                        document,
                        content,
                        students.get(i),
                        x,
                        y
                );
            }
        }

        document.save(outputFile);
    }
}
```

# 10. Final Version

```java
public void generateA4(
        List<Student> students,
        File outputFile)
        throws IOException {

    try (PDDocument document = new PDDocument()) {

        PDPage page =
                new PDPage(PDRectangle.A4);

        document.addPage(page);

        try (PDPageContentStream content =
                     new PDPageContentStream(
                             document,
                             page
                     )) {

            int maxCards =
                    Math.min(
                            students.size(),
                            4
                    );

            for (int i = 0;
                 i < maxCards;
                 i++) {

                int column = i % 2;
                int row = i / 2;

                float x =
                        50 + column * 250;

                float y =
                        600 - row * 200;

                drawCard(
                        document,
                        content,
                        students.get(i),
                        x,
                        y
                );
            }
        }

        document.save(outputFile);
    }
}
```

# 13. NetBeans Drag & Drop အတွက် Button Logic

`btnGenerateA4` ကို Drag & Drop ထည့်ပြီး double-click—

```java
private void btnGenerateA4ActionPerformed(
        java.awt.event.ActionEvent evt) {

    try {

        List<Student> students =
                getSelectedStudents();

        if (students.isEmpty()) {

            JOptionPane.showMessageDialog(
                    this,
                    "Please select students."
            );

            return;
        }

        JFileChooser chooser =
                new JFileChooser();

        chooser.setSelectedFile(
                new File("Student_ID_Cards.pdf")
        );

        int result =
                chooser.showSaveDialog(this);

        if (result ==
                JFileChooser.APPROVE_OPTION) {

            File outputFile =
                    chooser.getSelectedFile();

            StudentIDCardPDFService service =
                    new StudentIDCardPDFService();

            service.generateA4(
                    students,
                    outputFile
            );

            JOptionPane.showMessageDialog(
                    this,
                    "PDF generated successfully."
            );
        }

    } catch (Exception e) {

        e.printStackTrace();

        JOptionPane.showMessageDialog(
                this,
                "PDF generation failed:\n"
                        + e.getMessage()
        );
    }
}
```

private List<Student> getSelectedStudents() {

    List<Student> students =
            new ArrayList<>();

    int[] rows =
            tblStudents.getSelectedRows();

    for (int row : rows) {

        String studentId =
                tblStudents.getValueAt(
                        row,
                        0
                ).toString();

                studentDAO.findById(

                );
