# 2. `ChartPdfExporter.java`

Package:

```text
com.ams.report
```

Class:

```java
package com.ams.report;

import com.orsonpdf.PDFDocument;
import com.orsonpdf.Page;
import com.orsonpdf.PDFGraphics2D;

import org.jfree.chart.JFreeChart;

import java.awt.Rectangle;
import java.io.File;
import java.io.FileOutputStream;
import java.io.IOException;

public class ChartPdfExporter {

    public static void export(
            JFreeChart chart,
            File outputFile
    ) throws IOException {

        // A4 Landscape
        int width = 842;
        int height = 595;

        // Create PDF document
        PDFDocument document = new PDFDocument();

        // Create PDF page
        Rectangle bounds =
                new Rectangle(width, height);

        Page page =
                document.createPage(bounds);

        // Get PDF Graphics2D
        PDFGraphics2D graphics =
                page.getGraphics2D();

        // Draw JFreeChart
        chart.draw(
                graphics,
                bounds
        );

        // Get PDF bytes
        byte[] pdfBytes =
                document.getPDFBytes();

        // Write PDF file
        try (FileOutputStream output =
                     new FileOutputStream(outputFile)) {

            output.write(pdfBytes);
        }
    }
}
```

    graphics,

# 11. Exporter ကို final version ပြင်မယ်

```java
package com.ams.report;

import com.orsonpdf.PDFDocument;
import com.orsonpdf.Page;
import com.orsonpdf.PDFGraphics2D;

import org.jfree.chart.JFreeChart;

import java.awt.Rectangle;
import java.io.File;
import java.io.FileOutputStream;
import java.io.IOException;

public class ChartPdfExporter {

    private static final int PAGE_WIDTH = 842;
    private static final int PAGE_HEIGHT = 595;

    private static final int MARGIN = 40;

    private ChartPdfExporter() {
        // Utility class
    }

    public static void export(
            JFreeChart chart,
            File outputFile
    ) throws IOException {

        if (chart == null) {
            throw new IllegalArgumentException(
                    "Chart cannot be null."
            );
        }

        if (outputFile == null) {
            throw new IllegalArgumentException(
                    "Output file cannot be null."
            );
        }

        PDFDocument document =
                new PDFDocument();

        Rectangle pageBounds =
                new Rectangle(
                    PAGE_WIDTH,
                    PAGE_HEIGHT
                );

        Page page =
                document.createPage(
                    pageBounds
                );

        PDFGraphics2D graphics =
                page.getGraphics2D();

        Rectangle chartArea =
                new Rectangle(
                    MARGIN,
                    MARGIN,
                    PAGE_WIDTH - (MARGIN * 2),
                    PAGE_HEIGHT - (MARGIN * 2)
                );

        chart.draw(
                graphics,
                chartArea
        );

        byte[] pdfBytes =
                document.getPDFBytes();

        try (FileOutputStream output =
                     new FileOutputStream(
                         outputFile
                     )) {

            output.write(pdfBytes);
        }
    }
}
```
